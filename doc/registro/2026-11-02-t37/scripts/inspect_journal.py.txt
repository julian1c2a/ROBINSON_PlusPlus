import json, sys

path = sys.argv[1]
maxlen = int(sys.argv[2]) if len(sys.argv) > 2 else 300
with open(path, encoding='utf-8') as f:
    for i, l in enumerate(f):
        d = json.loads(l)
        keys = list(d.keys())
        t = d.get('type')
        s = json.dumps(d, ensure_ascii=False)
        print(i + 1, t, keys, len(s))
        print('   ', s[:maxlen].replace('\n', ' '))
