"""Imprime hallazgos de una ronda (JSON de workflow) con su arreglo propuesto
y la razon del verificador, recortados. Uso: ver_hallazgos.py <json> [ids,...] [n]"""
import json, sys

d = json.load(open(sys.argv[1], encoding='utf-8'))
want = set(sys.argv[2].split(',')) if len(sys.argv) > 2 and sys.argv[2] != '-' else None
n = int(sys.argv[3]) if len(sys.argv) > 3 else 500
res = d['result'] if isinstance(d, dict) else d
for lente in res:
    aud = lente['auditoria']
    vd = {v['id']: v for v in (lente.get('verificacion') or {}).get('veredictos', [])}
    for h in aud['hallazgos']:
        if want and h['id'] not in want:
            continue
        v = vd.get(h['id'], {})
        print('=' * 8, h['id'], '|', h.get('gravedad'), '->', v.get('gravedad_corregida'), '|', v.get('veredicto'))
        print('TIT:', h['titulo'])
        print('AFIRM:', (h.get('afirmacion') or '')[:n].replace('\n', ' '))
        print('ARREGLO:', (h.get('arreglo_propuesto') or '')[:n].replace('\n', ' '))
        print('VERIF:', (v.get('razon') or '')[:n].replace('\n', ' '))
