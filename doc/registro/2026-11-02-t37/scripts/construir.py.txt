"""T37: construye hallazgos.tsv y fuentes/ a partir de las fuentes (versionadas

import json, re, sys, os, hashlib, collections

RPP = sys.argv[1]
WF = sys.argv[2]
OUT = sys.argv[3]
FUENTES = os.path.join(OUT, 'fuentes')
os.makedirs(FUENTES, exist_ok=True)

H = re.compile(r'[‑‐‒–]')
def nh(s):
    return H.sub('-', s)

def rd(p):
    return open(os.path.join(RPP, p), encoding='utf-8').read()

def sha(x):
    if not isinstance(x, str):
        x = json.dumps(x, ensure_ascii=False, sort_keys=True)
    return hashlib.sha256(x.encode('utf-8')).hexdigest()[:16]

def journal(wf):
    started, res = {}, []
    for i, l in enumerate(open(os.path.join(WF, wf, 'journal.jsonl'), encoding='utf-8'), 1):
        d = json.loads(l)
        if d['type'] == 'started':
            started[d['key']] = (d['label'], d['agentId'])
        elif d['type'] in ('result', 'failed'):
            lab, aid = started.get(d['key'], ('?', d.get('agentId')))
            res.append({'linea': i, 'tipo': d['type'], 'label': lab, 'agentId': aid, 'result': d.get('result')})
    return res

def corto(s, n=170):
    s = ' '.join(str(s).split())
    return s if len(s) <= n else s[:n - 1] + '…'

# ---------------------------------------------------------------- ADRs que existen
ADRS = {}
for f in ['DECISIONS.md', 'doc/historia/DECISIONS-001-130.md']:
    for ln, l in enumerate(rd(f).split('\n'), 1):
        m = re.match(r'^## ADR[-‑](\d{3})', l)
        if m:
            ADRS['ADR-' + m.group(1)] = '%s:%d' % (f, ln)

# ---------------------------------------------------------------- Estado de los T
AUD = 'doc/AUDITORIA-SUELO-2026-10-12.md'
T = {}
cur = None
for ln, l in enumerate(rd(AUD).split('\n'), 1):
    m = re.match(r'^#### (T\d+[a-z]?) · ', l)
    if m:
        cur = m.group(1); T[cur] = {'linea': ln, 'estados': []}; continue
    if l.startswith('## '):
        cur = None
    if cur and l.startswith('*Estado'):
        T[cur]['estados'].append((ln, nh(l)))
ORD = {'tarde': 2, 'noche': 3}
def clave(e):
    ln, l = e
    m = re.match(r'\*Estado el (\d{4}-\d{2}-\d{2})(?: \(([^)]*)\))?', l)
    f = m.group(1) if m else '0'
    x = (m.group(2) or '') if m else ''
    fr = max([v for k, v in ORD.items() if k in x] or [0]) + (0.5 if ('al final' in x or 'después' in x) else 0)
    return (f, fr, '✅' in l)
TABIERTO = {'T10': 'parcial: la política hecha (ADR-141); falta separar procedencia/evidencia',
            'T18': 'abierto', 'T30': 'abierto', 'T32': 'parcial (🟡)', 'T37': 'abierto (este registro)',
            'T39': 'abierto', 'T48': 'abierto', 'T49': 'en espera (⏸, decisión del propietario)', 'T56': 'abierto',
            'T69': 'fuera de RPP (ADR-140)'}
def estado_t(t):
    d = T[t]
    if t in TABIERTO:
        return '%s ABIERTO: %s' % (t, TABIERTO[t])
    ln, l = max(d['estados'], key=clave)
    adrs = list(dict.fromkeys(re.findall(r'ADR-(\d{3})', l)))
    icon = '✅' if '✅' in l else ('🏁' if '🏁' in l else '?')
    return '%s %s ADR-%s, su Estado en :%d' % (t, icon, adrs[0] if adrs else '?', ln)
def ev_t(ts):
    return '; '.join('%s:%d (%s)' % (AUD, T[t]['linea'], estado_t(t)) for t in ts)

rows = []   # cada fila: dict
def fila(id_, origen, resumen, veredicto, estado, donde, evidencia, nota=''):
    assert estado in ('hecho', 'pendiente', 'descartado', 'sustituido'), (id_, estado)
    rows.append({'id': id_, 'origen': origen, 'resumen': resumen, 'veredicto': veredicto,
                 'estado': estado, 'donde': donde, 'evidencia': evidencia, 'nota': nota})

def A(n):
    """ref a un ADR, con su linea"""
    return '%s (%s)' % (n, ADRS[n])

def cargar_ronda(path):
    d = json.load(open(os.path.join(RPP, path), encoding='utf-8'))
    out = collections.OrderedDict()
    for lente in d['result']:
        vd = {v['id']: v for v in lente['verificacion']['veredictos']}
        for h in lente['auditoria']['hallazgos']:
            v = vd.get(h['id'], {})
            out[h['id']] = {'titulo': h['titulo'], 'gravedad': h.get('gravedad'), 'lente': lente['lente'],
                            'veredicto': v.get('veredicto'), 'gc': v.get('gravedad_corregida')}
    return out
