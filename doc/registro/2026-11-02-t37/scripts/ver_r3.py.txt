import json, sys
d = json.load(open(sys.argv[1], encoding='utf-8'))
want = set(sys.argv[2].split(','))
n = int(sys.argv[3]) if len(sys.argv) > 3 else 600
for lente in d:
    vd = {v['id']: v for v in lente['verificacion']['veredictos']}
    for h in lente['auditoria']['hallazgos']:
        if h['id'] not in want:
            continue
        print('=' * 6, h['id'], h['titulo'])
        keys = [k for k in h.keys() if k not in ('id', 'titulo')]
        for k in keys:
            v = h[k]
            if isinstance(v, list):
                v = ' | '.join(map(str, v))
            print('  %s: %s' % (k, str(v)[:n].replace('\n', ' ')))
        print('  VERIF:', (vd.get(h['id'], {}).get('razon') or '')[:n].replace('\n', ' '))
