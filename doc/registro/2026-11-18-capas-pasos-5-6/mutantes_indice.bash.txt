corre() {  # NOMBRE PATRÓN-ESPERADO
  local sal rc
  sal=$(bash $M 2>&1); rc=$?
  if [ $rc -ne 0 ] && grep -q "❌.*$2" <<< "$sal"; then echo "✓ $1: rojo ($(grep -m1 "❌.*$2" <<< "$sal" | sed 's/^ *//' | cut -c1-150))"
  else echo "✗ $1: rc=$rc"; grep '❌' <<< "$sal" | head -5; fi
}
echo "## (a) la medida de uso, sin los usos del VALOR"

grep -c 'for u in ci.type.getUsedConstants do' $M

echo "## (b) la medida de uso, sin los usos del TIPO"
sed 's/for u in ci.type.getUsedConstants ++ ((val.map (·.getUsedConstants)).getD #\[\]) do/for u in ((val.map (·.getUsedConstants)).getD #[]) do/' check-indice.bash > $M
grep -c 'for u in ((val.map' $M

grep -v '^    if (cu == cm) next$' check-indice.bash > $M
grep -c 'if (cu == cm) next' $M
