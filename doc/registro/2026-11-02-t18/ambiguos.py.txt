# ambiguos.py OUT FICHERO_LEAN — lista los términos AMBIGUOS de una salida de lean y propone el candidato.
# Imprime TSV: linea, col, nombre, eleccion, regla, forma, candidatos (separados por « | »).
# Formas que reconoce (las tres son la misma clase: un identificador con varias interpretaciones):
#   A  «error: Ambiguous term» (nombre en la línea siguiente; candidatos tras «Possible interpretations:»)
#   Q  «error: ambiguous term, use fully qualified name, possible interpretations [X, Y]» (unfold, simp…)
#   P  «error: ambiguous, possible interpretations» (varias interpretaciones elaboran)
#   N  un «L:C Ambiguous term» ANIDADO dentro de un «error: overloaded, errors» (el de fuera no se toca aquí)
# Reglas (T18): (1) si exactamente un candidato es del propio sondeo (no empieza por ROBINSON_PlusPlus. ni FOL.)
# y el sondeo lo declara, ése (el sondeo compilaba al escribirse, así que el homónimo del build no estaba a la
# vista: entró después); (2) si son el de Minimal.Axioms y su duplicado de PremsOfDotPrf/PremsBdAllPrf
# (liftTerm_numeralM, liftTerm_nil, substTerm_nil: el mismo enunciado; los 28 sondeos son anteriores al
# 2026-09-10, cuando entraron los duplicados), el de Minimal.Axioms; (3) si no, «?» (a mano).
import re, sys

out = open(sys.argv[1], encoding='utf-8', errors='replace').read().splitlines()
srcl = open(sys.argv[2], encoding='utf-8').read().split('\n')
src = '\n'.join(srcl)
pos_re = re.compile(r'^sondeos[\\/][^:]+:(\d+):(\d+): (error|warning|info)')
idre = re.compile(r"[A-Za-z_À-ɏͰ-Ͽ][A-Za-z0-9_'!?.À-ɏͰ-Ͽ₀-ₜ]*")

def nombre_en(ln, col):
    m = idre.match(srcl[ln - 1], col)
    return m.group(0) if m else '?'

def elige(cands):
    locales = [c for c in cands if not c.startswith('ROBINSON_PlusPlus.') and not c.startswith('FOL.')]
    if len(locales) == 1:
        corto = locales[0].split('.')[-1]
        if re.search(r'\b(def|theorem|lemma|abbrev|structure|inductive|instance)\s+' + re.escape(corto) + r'\b', src):
            return locales[0], 'local'
    if len(cands) == 2:
        ax = [c for c in cands if c.startswith('ROBINSON_PlusPlus.Minimal.Axioms.')]
        dup = [c for c in cands if c.startswith('ROBINSON_PlusPlus.Meta.PremsOfDotPrf.') or c.startswith('ROBINSON_PlusPlus.Meta.PremsBdAllPrf.')]
        if len(ax) == 1 and len(dup) == 1 and ax[0].split('.')[-1] == dup[0].split('.')[-1] \
           and ax[0].split('.')[-1] in ('liftTerm_numeralM', 'liftTerm_nil', 'substTerm_nil'):
            return ax[0], 'Minimal.Axioms'
    return '?', 'a mano'

def cands_indent(block, start, ind):
    """candidatos: líneas con EXACTAMENTE `ind` espacios de sangría tras «Possible interpretations:»"""
    cs = []
    for l in block[start:]:
        if l.strip() == '':
            continue
        k = len(l) - len(l.lstrip(' '))
        if k < ind:
            break
        if k == ind:
            cs.append(l.strip().split()[0])
    return cs

sitios = {}
i = 0
while i < len(out):
    m = pos_re.match(out[i])
    if not m:
        i += 1; continue
    ln, col = int(m.group(1)), int(m.group(2))
    j = i + 1
    while j < len(out) and not pos_re.match(out[j]) and (out[j].strip() == '' or out[j].startswith(' ')):
        j += 1
    block = out[i:j]
    head = block[0]
    if head.rstrip().endswith('error: Ambiguous term'):
        name = block[1].strip().split()[0] if len(block) > 1 else '?'
        # «Possible interpretations:» va sin sangría: el bloque se cortó ahí; recógelo a mano (dos líneas más abajo
        # como mucho: nunca el de otro error)
        cs = []
        t = i + 1
        while t < min(i + 4, len(out)) and not out[t].startswith('Possible interpretations'):
            t += 1
        if t < min(i + 4, len(out)):
            u = t + 1
            while u < len(out) and not pos_re.match(out[u]) and (out[u].strip() == '' or out[u].startswith(' ')):
                u += 1
            cs = cands_indent(out[t + 1:u], 0, 2)
            j = max(j, u)
        sitios.setdefault((ln, col), (name, 'A', cs))
    elif 'ambiguous term, use fully qualified name, possible interpretations' in head:
        txt = ' '.join(block)
        mm = re.search(r'possible interpretations \[(.*?)\]', txt)
        cs = [c.strip() for c in mm.group(1).split(',')] if mm else []
        sitios.setdefault((ln, col), (nombre_en(ln, col), 'Q', cs))
    elif 'ambiguous, possible interpretations' in head:
        cs = cands_indent(block, 1, 2)
        sitios.setdefault((ln, col), (nombre_en(ln, col), 'P', cs))
    elif 'overloaded, errors' in head:
        for t, l in enumerate(block):
            mm = re.match(r'^  (\d+):(\d+) Ambiguous term', l)
            if mm:
                nl, nc = int(mm.group(1)), int(mm.group(2))
                name = block[t + 1].strip().split()[0] if t + 1 < len(block) else '?'
                pi = next((u for u in range(t, len(block)) if block[u].strip().startswith('Possible interpretations')), None)
                cs = cands_indent(block, pi + 1, 4) if pi is not None else []
                sitios.setdefault((nl, nc), (name, 'N', cs))
    i = j

for (ln, col), (name, forma, cs) in sorted(sitios.items()):
    e, r = elige(cs)
    print(f'{ln}\t{col}\t{name}\t{e}\t{r}\t{forma}\t{" | ".join(cs)}')
