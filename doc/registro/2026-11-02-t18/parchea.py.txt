# parchea.py FICHERO EDICIONES.tsv — cualifica identificadores en las posiciones dadas (las de lean: línea desde 1,
# columna en puntos de código desde 0). Cada línea de EDICIONES: linea<TAB>col<TAB>nombre<TAB>reemplazo[...].
# Comprueba que en esa posición empieza EXACTAMENTE el identificador `nombre` (no un trozo de otro) y aplica de
# abajo arriba y de derecha a izquierda. Sale 1 sin escribir nada si alguna posición no casa.
import sys, re

f, ed = sys.argv[1], sys.argv[2]
txt = open(f, encoding='utf-8', newline='').read()
lines = txt.split('\n')
edits = []
for l in open(ed, encoding='utf-8'):
    if not l.strip() or l.startswith('#'):
        continue
    p = l.rstrip('\n').split('\t')
    edits.append((int(p[0]), int(p[1]), p[2].split()[0], p[3]))
idch = re.compile(r"[A-Za-z0-9_'!?.À-ɏͰ-Ͽᴀ-ᶿ₀-ₜ]")
ok = True
for ln, col, name, rep in edits:
    s = lines[ln - 1]
    if s[col:col + len(name)] != name:
        print(f'NO CASA {ln}:{col}: esperaba {name!r}, hay {s[col:col + len(name) + 10]!r}'); ok = False; continue
    after = s[col + len(name):col + len(name) + 1]
    before = s[col - 1:col] if col > 0 else ''
    if (after and idch.match(after)) or (before and idch.match(before)):
        print(f'NO CASA {ln}:{col}: {name!r} es parte de otro identificador: {s[max(0,col-5):col + len(name) + 5]!r}'); ok = False
if not ok:
    sys.exit(1)
seen = set()
for ln, col, name, rep in sorted(edits, key=lambda e: (e[0], e[1]), reverse=True):
    if (ln, col) in seen:
        continue
    seen.add((ln, col))
    s = lines[ln - 1]
    lines[ln - 1] = s[:col] + rep + s[col + len(name):]
open(f, 'w', encoding='utf-8', newline='').write('\n'.join(lines))
print(f'{len(seen)} sitios cualificados en {f}')
