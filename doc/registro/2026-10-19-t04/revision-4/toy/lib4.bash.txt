nuevo_repo() {   # $1 = directorio (se borra y se crea): repo con la tabla inicial creada y la rama `verificado`
  rm -rf "$1"; mkdir -p "$1"; cd "$1" || exit 2
  git init -q . && git config user.email t@t && git config user.name t && git config core.autocrlf false
  git checkout -q -b master 2>/dev/null || true
  cp "$ORIG/b-citas.awk" "$ORIG/b-control.bash" .
  printf '%s\n' vivo_ok > vivos; : > env; : > sond; : > mod; : > rut
  { echo 'theorem raa : True := trivial'; for i in $(seq 1000 2000); do echo "theorem nombre_$i : True := trivial"; done; } > X.lean
  printf '## ADR-900: sube `muerto_uno` (un ADR viejo)\n\n## ADR-901: otra cosa\n' > DECISIONS.md
  printf '%s\n' "$DOC1" "$DOC2" > doc.md
  printf 'salida.txt\n' > .gitignore
  DOCS="doc.md"; BASE=verificado
  commit base; git branch -f verificado HEAD; PRE_TABLA=$(git rev-parse HEAD)
  corre 2 "" > /dev/null; commit inicial; git branch -f verificado HEAD; INICIAL=$(git rev-parse HEAD)
}
corre() {   # $1 = modo  $2 = B_SUBE_ADR   (BASE vacío ⇒ B_BASE_REF sin poner: la omisión, origin/master)
  ( B_DECLS=vivos B_ENV=env B_SOND=sond B_MOD=mod B_RUT=rut B_GITS=. B_DOCS_LISTA=$DOCS
    if [ -n "$BASE" ]; then B_BASE_REF=$BASE; else unset B_BASE_REF; fi
    B_DEUDA_MODO=$1 B_SUBE_ADR=$2
    b_adr_texto() { awk -v a="## $1:" '/^## / { en = (index($0, a) == 1) } en' DECISIONS.md; }
    b_adr_texto_base() { ${ADR_BASE_FN:-b_adr_base_rpp} "$1"; }
    b_adr_base_rpp() { git show "$B_BASE:DECISIONS.md" 2>/dev/null | awk -v a="## $1:" '/^## / { en = (index($0, a) == 1) } en'; }
    b_adr_base_fol() { :; }
    source ./b-control.bash > salida.txt 2>&1; echo "FAIL=$B_FAIL MEDIDO=$B_MEDIDO"
    grep -E '❌|✓|⚠️|techo:|deuda declarada' salida.txt | head -8 )
}
paso() {   # $1 = descripción  $2 = modo  [$3 = B_SUBE_ADR]
  echo "── $1 (modo $2${3:+, B_SUBE_ADR=$3}${BASE:+, B_BASE_REF=$BASE})"; corre "$2" "${3:-}" | sed 's/^/      /'
}
