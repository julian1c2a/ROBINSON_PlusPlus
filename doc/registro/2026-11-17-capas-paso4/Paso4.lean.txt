import ROBINSON_PlusPlus
/-! B1 (ADR-266), paso 4: (a) los cortes —lo que se QUEDA y usa algo que se va; de qué módulos depende lo que se va—;
(b) de qué módulos del proyecto depende lo que se queda (los imports que le hacen falta); (c) las declaraciones con un
espacio de nombres AJENO a su módulo (NAMING-CONVENTIONS §6). Fuera del árbol: `lake env lean .lake/tmp/Paso4.lean`. -/
open Lean Elab Command

def modDe (env : Environment) (n : Name) : Option Name := do
  let i ← env.getModuleIdxFor? n
  env.header.moduleNames[i.toNat]?

def usadas (env : Environment) (n : Name) : Array Name := Id.run do
  let some ci := env.find? n | return #[]
  let mut s := ci.type.getUsedConstants
  if let some v := ci.value? (allowOpaque := true) then s := s ++ v.getUsedConstants
  return s

/-- el nombre «de usuario» de una constante auxiliar (`f.eq_1`, `f._proof_1`, `f.match_1` → `f`) -/
def raiz (n : Name) : Name := Id.run do
  let mut m := n
  while m.isInternal || (match m with | .str _ s => s.startsWith "eq_" || s.startsWith "match_" || s == "eq_def" || s.startsWith "_" | _ => false) do
    m := m.getPrefix
  return m

elab "#corte " m:ident " | " xs:ident* : command => do
  let env ← getEnv
  let mod := m.getId
  let some idx := env.getModuleIdx? mod | throwError "no existe el módulo {mod}"
  let vanNames : Array Name := xs.map (fun x => `ROBINSON_PlusPlus.Meta ++ x.getId)
  for v in vanNames do unless env.contains v do throwError "no existe {v}"
  let van : NameSet := vanNames.foldl (·.insert ·) {}
  let mut quedanMal : Array String := #[]
  let mut modsVan : Std.HashMap Name Nat := {}
  let mut modsQuedan : Std.HashMap Name (Array String) := {}
  for (n, _) in env.constants.toList do
    unless env.getModuleIdxFor? n == some idx do continue
    let r := raiz n
    let seVa := van.contains r
    for d in usadas env n do
      let rd := raiz d
      if !seVa && van.contains rd && !n.isInternal then
        quedanMal := quedanMal.push s!"{n} usa {rd}"
      if let some md := modDe env d then
        if md != mod && ((`ROBINSON_PlusPlus).isPrefixOf md || (`FOL).isPrefixOf md) then
          if seVa then
            modsVan := modsVan.insert md ((modsVan.getD md 0) + 1)
          else if !n.isInternal then
            modsQuedan := modsQuedan.insert md (((modsQuedan.get? md).getD #[]).push s!"{n} → {d}")
  let mods := modsVan.toArray.qsort (fun a b => a.1.toString < b.1.toString)
  let qs := modsQuedan.toArray.qsort (fun a b => a.1.toString < b.1.toString)
  let lq := qs.toList.map fun (md, ejs) => s!"    {md} ({ejs.size}), p. ej. {ejs[0]!}"
  logInfo m!"{mod} — se van {vanNames.size}.\n  (a) lo que se QUEDA y usa algo que se va: {if quedanMal.isEmpty then "nada" else "\n    " ++ "\n    ".intercalate quedanMal.toList}\n  (b) lo que se va depende de: {", ".intercalate (mods.toList.map fun (md, k) => s!"{md} ({k})")}\n  (c) lo que se QUEDA depende de:\n{"\n".intercalate lq}"

/-- las declaraciones (no internas) de cada módulo del proyecto cuyo nombre no empieza por el del módulo -/
elab "#ajenos" : command => do
  let env ← getEnv
  let mut porMod : Std.HashMap Name (Array Name) := {}
  for (n, _) in env.constants.toList do
    if n.isInternal then continue
    let some md := modDe env n | continue
    unless (`ROBINSON_PlusPlus).isPrefixOf md do continue
    if md.isPrefixOf n then continue
    porMod := porMod.insert md (((porMod.get? md).getD #[]).push n)
  let filas := porMod.toArray.qsort (fun a b => a.1.toString < b.1.toString)
  let total := filas.foldl (fun k (_, ns) => k + ns.size) 0
  let lineas := filas.toList.map fun (md, ns) =>
    let ns' := ns.qsort (·.toString < ·.toString)
    s!"  {md} ({ns.size}):\n      {"\n      ".intercalate (ns'.toList.map toString)}"
  logInfo m!"declaraciones con un espacio de nombres AJENO a su módulo: {total}, en {filas.size} módulos\n{"\n".intercalate lineas}"

#ajenos
