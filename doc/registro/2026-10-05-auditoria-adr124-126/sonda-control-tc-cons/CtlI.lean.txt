set_option autoImplicit false

open FOL FOL.Metamath.Semantics ROBINSON_PlusPlus.Minimal.Axioms

namespace ControlNeg

theorem tcFnN_succ (n : Nat) :
    tcFnN (n + 1) = consN 1 (consN (codeNatStr succ_sym) (consN (consN (tcFnN n) 0) 0)) := rfl

/-- La forma que `ax_tc_cons` le pide a `tcFn (cons a b)` —un símbolo y DOS argumentos— no es la de `tcFnN`
    sobre un sucesor —un argumento—, para TODO `a`, `b` (las variables impiden evaluar nada). -/
theorem tcFnN_consN_ne (a b X : Nat) :
    tcFnN (consN a b) ≠ consN 1 (consN X (consN (consN (tcFnN a) (consN (tcFnN b) 0)) 0)) := by
  intro h
  -- ⛔ ni un `have … := h` por defeq: el unificador desplegaría `consN` sobre `codeNatStr succ_sym` y evaluaría
  --    triangulares de cientos de miles (medido: >3 GB en 14 s). Sólo `rw` con lemas ∀.
  rw [show consN a b = triN (a + b) + b + 1 from rfl, tcFnN_succ] at h
  have h2 := (ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_inj h).2
  have h3 := (ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_inj h2).2
  have h4 := (ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_inj h3).1
  have h5 := (ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_inj h4).2
  exact ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_ne_zero _ _ h5.symm

/-- ⛔ Control negativo: `MNV V` REFUTA `ax_tc_cons` (el axioma que hizo inconsistente la teoría, ADR‑012). -/
theorem control_tc_cons {V : Nat} : ¬ ∀ v : Nat → Nat, evalFormula (MNV V) v ax_tc_cons := by
  intro h
  have h0 := h (fun _ => 0) 0 0
  have h1 : evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (tcFn (cons (.var 1) (.var 0)))
      = evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (cons (numeralM 1) (cons (strCodeM cons_sym)
          (cons (cons (tcFn (.var 1)) (cons (tcFn (.var 0)) nil)) nil))) := h0
  simp only [ev_tcFn, ev_cons, ev_nil, ev_numeralM, ev_strCodeM] at h1
  exact tcFnN_consN_ne _ _ _ h1

end ControlNeg
