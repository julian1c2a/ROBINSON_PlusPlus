theorem tcFnN_succ (n : Nat) :
    tcFnN (n + 1) = consN 1 (consN (codeNatStr succ_sym) (consN (consN (tcFnN n) 0) 0)) := rfl

/-- ⛔ Control negativo: `MNV V` REFUTA `ax_tc_cons` (el axioma que hizo inconsistente la teoría, ADR‑012). -/
theorem control_tc_cons {V : Nat} : ¬ ∀ v : Nat → Nat, evalFormula (MNV V) v ax_tc_cons := by
  intro h
  have h0 := h (fun _ => 0) 0 0
  have h1 : evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (tcFn (cons (.var 1) (.var 0)))
      = evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (cons (numeralM 1) (cons (strCodeM cons_sym)
          (cons (cons (tcFn (.var 1)) (cons (tcFn (.var 0)) nil)) nil))) := h0
  have e1 : evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (.var 1) = 0 := rfl
  have e0 : evalTerm (MNV V) (shiftEnv (shiftEnv (fun _ => 0) 0) 0) (.var 0) = 0 := rfl
  simp only [ev_tcFn, ev_cons, ev_nil, ev_numeralM, ev_strCodeM, e1, e0] at h1
  rw [show consN 0 0 = 0 + 1 from rfl, tcFnN_succ] at h1
  have h2 := (ROBINSON_PlusPlus.Meta.CodeNatInjPrf.consN_inj h1).2
  sorry

end ControlNeg
