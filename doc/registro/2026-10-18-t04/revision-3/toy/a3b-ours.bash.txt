# y las tres citas), la fusión es TREESAME a su primer padre (no la lista git log -- tabla) y es un avance rápido de master.

paso "(C1) tabla inicial: 3" 2; commit C1; C1=$(git rev-parse HEAD)
printf '%s\n' 'Uno: `muerto_dos`.' 'Dos: arreglado.' 'Tres: arreglado.' > doc.md
paso "(C2) --b-deuda: 1" 1; commit C2; C2=$(git rev-parse HEAD)
