# p_proto.bash (el prototipo del cruce, antes del bloque: sólo FOL/ y TheoryFramework/)
#!/bin/bash
# prototipo del cruce de footprints en prosa (T39): FOL/ y TheoryFramework/ contra las filas de FOL de la TABLA de
# ../ROBINSON_PlusPlus/check-footprints.bash
export PATH="/usr/bin:$PATH"
cd /e/dropbox/github/lean4/FOL || exit 2
RPP=../ROBINSON_PlusPlus
FILAS=$(mktemp)
awk "/^read -r -d '' TABLA <<'EOF'/{f=1;next} f&&/^EOF\$/{exit} f" "$RPP/check-footprints.bash" > "$FILAS"
echo "filas en la TABLA: $(grep -c . "$FILAS")"
find FOL TheoryFramework -name '*.lean' | LC_ALL=C sort | xargs env LC_ALL=C awk -v FILAS="$FILAS" '
function norm(s,   n, z, i, out, t, j, k) {
  gsub(/[][` ]/, "", s); n = split(s, z, ",")
  for (i = 1; i <= n; i++) for (j = i + 1; j <= n; j++) if (z[j] < z[i]) { t = z[i]; z[i] = z[j]; z[j] = t }
  out = ""; for (i = 1; i <= n; i++) if (z[i] != "") out = out (out == "" ? "" : ",") z[i]
  return out == "" ? "-" : out }
BEGIN {
  while ((getline l < FILAS) > 0) { split(l, p, "|"); nm = p[1]; fp = norm(p[2] == "-" ? "" : p[2])
    full[nm] = fp; k = nm; while (sub(/^[^.]*[.]/, "", k)) { if (k in suf && suf[k] != nm) amb[k] = 1; suf[k] = nm } }
  FP = "\\[(propext|Quot[.]sound|Classical[.]choice)(, *(propext|Quot[.]sound|Classical[.]choice))*\\]" }
{ linea = $0; resto = linea; base = 0
  while (match(resto, FP)) {
    ms = RSTART; ml = RLENGTH; fp = norm(substr(resto, ms, ml)); antes = substr(linea, 1, base + ms - 1)
    # el tramo: desde el último «;», «. », «—» o «(» antes del footprint
    seg = antes; while (match(seg, /(; |[.] |— |\()/)) seg = substr(seg, RSTART + RLENGTH)
    s2 = seg
    while (match(s2, /`[A-Za-z_][A-Za-z0-9_.'"'"'₀₁₂₃ᵢ]*`/)) {
      id = substr(s2, RSTART + 1, RLENGTH - 2); s2 = substr(s2, RSTART + RLENGTH)
      nm = (id in full) ? id : ((id in suf && !(id in amb)) ? suf[id] : "")
      if (nm == "") continue
      vistos++
      if (full[nm] != fp) print FILENAME ":" FNR ": `" id "` dice " fp " · la fila (" nm ") dice " full[nm] "\t" substr(linea, 1, 150)
    }
    base += ms + ml - 1; resto = substr(resto, ms + ml)
  }
}
END { print "pares (cita, fila) comparados: " vistos + 0 > "/dev/stderr" }'
rm -f "$FILAS"
