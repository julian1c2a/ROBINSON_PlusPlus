/-- En un modelo estándar donde Prov⌜G⌝ vale, G es falso: Prf ⊬ G, sin Gödel. -/
theorem no_prf_godelCN_por_modelo (hM : Estandar M)
    (hProv : evalFormula M (fun _ => 0) (provCodeC' godelCN)) : ¬ Prf godelCN := by
  intro hG
  have hfp := prf_sound hM prf_godelCN_fixedpoint (fun _ => 0)
  have hg := prf_sound hM hG (fun _ => 0)
  simp only [_root_.iff, neg, evalFormula] at hfp
  exact hfp.1 hg hProv

/-- Ídem Gödel II. -/
theorem no_prf_con_por_modelo (hM : Estandar M)
    (hProv : evalFormula M (fun _ => 0) (provCodeC' Formula.bottom)) :
    ¬ Prf consistencyFormula' := by
  intro hC
  have hc := prf_sound hM hC (fun _ => 0)
  simp only [consistencyFormula', neg, evalFormula] at hc
  exact hc hProv
