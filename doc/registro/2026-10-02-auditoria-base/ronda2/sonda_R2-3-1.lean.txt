  simp only [axNew, nPsi, psiT, diagTermM, substFormula, axiomsCodeT, concat, cons, nil, zero,

/-- (ii) delta = ⌜axNew⌝, con `prf_diag_arith_num` tal cual. -/
theorem prf_delta : Prf (substfc zero (tcFn nPsi) nPsi =eq formCodeM axNew) := by
  have h1 := prf_diag_arith_num psiT
  have hl : substTerm 0 (numeral (codeNat psiT)) diagTerm = substfc zero (tcFn nPsi) nPsi := by
    rw [nPsi, numeralM_eq]; rfl
  rw [hl, ← axNew_selfApp] at h1
  have h2 := prf_eq_symm (prf_formCode_numeral axNew)
  rw [← formCodeM_eq] at h2

    Prf (axiomsCodeT =eq listFormCodeM (axioms ++ [axNew])) :=
