/-- `x < σ0 ⟹ x = 0`, en cualquier contexto. -/
theorem prfH_eq_zero_of_lt_one {Γ : List Formula} {x : Term}
    (h : PrfH Γ (lt x (succ zero))) : PrfH Γ (x =eq zero) :=
  PrfH_or_elim (PrfH.mp _ _ _ (prf_to_prfH (prf_le_of_lt_succ x zero) _) h)
    (PrfH.mp _ _ _ (PrfH.incl0 _ _ (Prfᵢ.efq _))
      (PrfH.mp _ _ _ (prf_to_prfH (prf_not_lt_zero x) _) (PrfH.hyp _ _ (List.Mem.head _))))
    (PrfH.hyp _ _ (List.Mem.head _))

  refine prf_or_elim (prf_nil_or_cons (succ zero)) (prf_succ_ne_zero zero) (prf_deduction ?_)

      (succ zero =eq cons (carc (succ zero)) (cdrc (succ zero))) := prfH_hyp_self _
  have hc := prfH_eq_zero_of_lt_one (PrfH_lt_subst2 (PrfH_eq_symm hE)
    (prf_to_prfH (prf_cantor_mono_left (carc (succ zero)) (cdrc (succ zero))) _))
  have hd := prfH_eq_zero_of_lt_one (PrfH_lt_subst2 (PrfH_eq_symm hE)
    (prf_to_prfH (prf_cantor_mono_right (carc (succ zero)) (cdrc (succ zero))) _))
  -- master: consN 0 0 = triN 1 + 1 = 2
  have hev : Prf (cons zero zero =eq succ (succ zero)) := prf_cons_eval 0 0
  have h12 : PrfH [consOk (succ zero)] (succ zero =eq succ (succ zero)) :=

      (PrfH_eq_trans (PrfH_congr_cons_head hc) (PrfH_congr_cons_tail hd)) (prf_to_prfH hev _))
  exact PrfH.mp _ _ _ (prf_to_prfH (prf_succ_ne_zero zero) _)
    (PrfH_eq_symm (PrfH.mp _ _ _ (prf_to_prfH (prf_succ_inj zero (succ zero)) _) h12))
