/-- Sin cuantificadores, con todos los átomos de relación en `D` (la igualdad, siempre). -/
def qfD (D : List Char → Bool) : Formula → Bool
  | .bottom => true
  | .atom p _ => D p
  | .eq _ _ => true
  | .impl A B => qfD D A && qfD D B
  | .forall _ => false
  | .and A B => qfD D A && qfD D B
  | .or A B => qfD D A && qfD D B
  | .ex _ => false

mutual
/-- P: `eval φ → eval (gg φ)`. -/
def canP (D : List Char → Bool) : Formula → Bool
  | .bottom => true
  | .atom _ _ => true
  | .eq _ _ => true
  | .impl A B => canN D A && canP D B
  | .forall A => canP D A
  | .and A B => canP D A && canP D B
  | .or A B => canP D A && canP D B
  | .ex A => canP D A
/-- N: `eval (gg φ) → ¬¬ eval φ`. -/
def canN (D : List Char → Bool) : Formula → Bool
  | .bottom => true
  | .atom _ _ => true
  | .eq _ _ => true
  | .impl A B => canP D A && canN D B
  | .forall A => canS D A
  | .and A B => canN D A && canN D B
  | .or A B => canN D A && canN D B
  | .ex A => canN D A
/-- S: `eval (gg φ) → eval φ`. -/
def canS (D : List Char → Bool) : Formula → Bool
  | .bottom => true
  | .atom p _ => D p
  | .eq _ _ => true
  | .impl A B => canP D A && canS D B
  | .forall A => canS D A
  | .and A B => canS D A && canS D B
  | .or A B => (qfD D A && canN D A && canS D B) || (qfD D B && canN D B && canS D A)
  | .ex _ => false
end

def esq : Formula → String
  | .bottom => "⊥"
  | .atom p _ => String.ofList p
  | .eq _ _ => "="
  | .impl A B => "(" ++ esq A ++ " ⇒ " ++ esq B ++ ")"
  | .forall A => "∀" ++ esq A
  | .and A B => "(" ++ esq A ++ " ∧ " ++ esq B ++ ")"
  | .or A B => "(" ++ esq A ++ " ∨ " ++ esq B ++ ")"
  | .ex A => "∃" ++ esq A

def todos : List Char → Bool := fun _ => true
def nada : List Char → Bool := fun _ => false
def ltMem : List Char → Bool := fun p => p == in_sym

def informe (D : List Char → Bool) : List String :=
  (axiomsBase.zipIdx).filterMap fun (a, i) =>
    if canP D a then none else some (toString i ++ ": " ++ esq a)
