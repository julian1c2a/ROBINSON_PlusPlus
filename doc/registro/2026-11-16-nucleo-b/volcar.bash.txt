#!/bin/bash
# Como doc/registro/2026-11-09-deuda/volcar-citas.bash.txt, pero RESTAURA b-control.bash desde una copia (no con git
# checkout: lleva cambios sin commit, los de A7). Vuelca la salida entera de b-citas.awk y sus entradas.
# Uso (Git Bash, desde PowerShell): bash volcar.bash [RPP|FOL|ambos]
S=/d/msys64/tmp/claude/d75c25be1c300eb1/a8ec29ce-df3a-4c41-9e7a-c91f87db9b4e/scratchpad/a7
export PATH=/c/Users/julia/.elan/bin:$PATH
QUE=${1:-ambos}
for R in ROBINSON_PlusPlus FOL; do
  case "$QUE" in RPP) [ "$R" = FOL ] && continue ;; FOL) [ "$R" = ROBINSON_PlusPlus ] && continue ;; esac
  cd /e/dropbox/github/lean4/$R || exit 2
  n=$(grep -n '^  b_cuenta < "$B_CIT" > "$B_ACT"$' b-control.bash | cut -d: -f1)
  [ -n "$n" ] || { echo "$R: no encuentro la línea del gancho"; exit 2; }
  E="$S/entradas-$R"; mkdir -p "$E"
  cp b-control.bash "$S/b-control.copia-$R"
  awk -v n="$n" -v out="$S/citas-$R.txt" -v e="$E" '{ print } NR == n {
      print "  cp \"$B_CIT\" \"" out "\""
      print "  cp \"$B_DECLS_A\" \"" e "/vivos\"; cp \"$B_ENV_A\" \"" e "/env\"; cp \"$B_SOND_A\" \"" e "/sond\""
      print "  cp \"$B_MOD_A\" \"" e "/mod\"; cp \"$B_RUT_A\" \"" e "/rut\"; cp \"$B_HIST\" \"" e "/hist\"" }' b-control.bash > "$S/b-control.tmp"
  cp "$S/b-control.tmp" b-control.bash
  bash check-doc-sync.bash > "$S/ds-$R.txt" 2>&1
  cp "$S/b-control.copia-$R" b-control.bash
  cmp -s b-control.bash "$S/b-control.copia-$R" && echo "$R: b-control.bash restaurado" || echo "$R: ⚠️ b-control.bash NO restaurado"
  echo "$R: $(grep -c '^B' "$S/citas-$R.txt") citas muertas; entradas: $(wc -l "$E"/* | tail -1)"
done
