def leer(p):
    return io.open(p, encoding='utf-8').read()


def escribir(p, t):
    io.open(p, 'w', encoding='utf-8', newline='\n').write(t)


def uno(t, s, quien):
    k = t.count(s)
    if k != 1:
        sys.exit(f'ANCLA {quien}: {k} apariciones de {s[:80]!r}')
    return t.index(s)


def cortar(t, ini, fin, quien):
    """Corta desde la línea que empieza en `ini` hasta el final de la línea de `fin` (incluida) y la línea en blanco
    que la siga. Devuelve (texto sin el bloque, bloque)."""
    a = uno(t, ini, quien + '/ini')
    b = t.index(fin, a)
    if t.count(fin, a) < 1:
        sys.exit(f'ANCLA {quien}/fin')
    b = t.index('\n', b + len(fin) - 1) + 1
    if t[b:b + 1] == '\n':
        b += 1
    return t[:a] + t[b:], t[a:b]


def reemplazar(t, viejo, nuevo, quien):
    uno(t, viejo, quien)
    return t.replace(viejo, nuevo)

fich = [os.path.join(d, f) for d0 in dirs for d, _, fs in os.walk(d0) for f in fs if f.endswith('.lean')]
fich = [f for f in fich if '/pasadas/' not in f.replace(os.sep, '/')]
fich += [R + f for f in os.listdir(R) if f.startswith('check-') and f.endswith('.bash')]
for f in fich:
    t = leer(f)
    t0 = t
    for v, n in ren.items():
        t = re.sub(re.escape(v) + r'(?![\w\'₀-₉])', n, t)
    for v, n in cortos.items():
        t = re.sub(r'(?<![\w.])' + re.escape(v) + r'(?![\w\'₀-₉])', n, t)
    if t != t0:
        escribir(f, t)
        toc[f.replace(R, '')] = sum(t0.count(v) for v in list(ren) + list(cortos))
for f, k in sorted(toc.items()):
    print('renombrado', f, k)
