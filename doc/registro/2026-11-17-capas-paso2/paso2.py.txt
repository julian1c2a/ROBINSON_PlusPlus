def lee(p):
    t = io.open(p, encoding='utf-8', newline='').read()
    nl = '\r\n' if '\r\n' in t else '\n'
    return t.replace('\r\n', '\n'), nl

src = {}   # módulo → [texto LF, nl]
def mod(m):
    if m not in src:
        src[m] = list(lee(M + m + '.lean'))
    return src[m]

def corta(m, ini, fin, unica=True):
    """quita de `m` las líneas desde la primera que EMPIEZA por `ini` hasta la anterior a la primera, después, que
    empieza por `fin`; devuelve el texto quitado (con su salto final). Con `unica`, `ini` tiene que salir una vez."""
    t = mod(m)[0]
    ls = t.split('\n')
    a = [i for i, l in enumerate(ls) if l.startswith(ini)]
    assert (len(a) == 1) if unica else a, (m, ini, len(a))
    a = a[0]
    b = [i for i, l in enumerate(ls) if i > a and l.startswith(fin)]
    assert b, (m, fin)
    b = b[0]
    bloque = '\n'.join(ls[a:b]) + '\n'
    mod(m)[0] = '\n'.join(ls[:a] + ls[b:])
    return bloque

def cambia(m, viejo, nuevo, n=1):
    t = mod(m)[0]
    assert t.count(viejo) == n, (m, viejo[:60], t.count(viejo))
    mod(m)[0] = t.replace(viejo, nuevo)

def quita_linea(m, linea):
    cambia(m, '\n' + linea + '\n', '\n')

NOTA_PUREZA = ('-- (✏️ 2026‑11‑17, ADR‑268: el control POSITIVO de este `#pureza` —lo que nombra la semántica— está en\n'
               '-- `Meta/Controles.lean`, en C8: aquí no se importa la semántica.)\n')
def pureza_sin_control(m, linea_control):
    cambia(m, '\n  | ' + linea_control + '\n', '\n  |\n')
    t = mod(m)[0]
    k = t.index('\n#pureza ')
    mod(m)[0] = t[:k + 1] + NOTA_PUREZA + t[k + 1:]
